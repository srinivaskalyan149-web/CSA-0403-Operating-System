#include<stdio.h>
#include<pthread.h>

pthread_mutex_t m;

void *task(void *p){
    pthread_mutex_lock(&m);
    printf("Critical Section\n");
    pthread_mutex_unlock(&m);
}
int main(){
    pthread_t t;
    pthread_mutex_init(&m,NULL);
    pthread_create(&t,NULL,task,NULL);
    pthread_join(t,NULL);
}
